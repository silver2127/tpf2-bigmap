cmake_minimum_required(VERSION 3.13)
project(tpf2_bigmap_linux LANGUAGES CXX ASM)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(tpf2_bigmap MODULE bigmap_linux.cpp terrain_minmax.cpp terrain_minmax.S terrain_copy.cpp terrain_copy.S)
set_target_properties(tpf2_bigmap PROPERTIES PREFIX "" SUFFIX ".so")
target_compile_options(tpf2_bigmap PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_compile_options(tpf2_bigmap PRIVATE -ffp-contract=off)
target_link_libraries(tpf2_bigmap PRIVATE Threads::Threads)
target_link_options(tpf2_bigmap PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_terrain_cache tests/terrain_cache_test.cpp)
  target_compile_options(test_terrain_cache PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(test_terrain_cache PRIVATE Threads::Threads)
  add_test(NAME terrain_cache COMMAND test_terrain_cache)
  foreach(kernel refine align)
    add_executable(terrain_${kernel}_oracle tests/terrain_${kernel}_oracle.cpp)
    target_compile_options(terrain_${kernel}_oracle PRIVATE -UNDEBUG -ffp-contract=off)
    target_link_libraries(terrain_${kernel}_oracle PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  endforeach()
  add_executable(test_bigmap tests/bigmap_test.cpp tests/town_stub.S terrain_minmax.cpp terrain_minmax.S terrain_copy.cpp terrain_copy.S)
  target_compile_options(test_bigmap PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(test_bigmap PRIVATE Threads::Threads)
  add_test(NAME bigmap COMMAND test_bigmap)
  add_executable(test_terrain_copy tests/terrain_copy_test.cpp tests/copy_stub.S terrain_copy.cpp terrain_copy.S)
  target_compile_options(test_terrain_copy PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME terrain_copy COMMAND test_terrain_copy)
endif()
